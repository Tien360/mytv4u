<script>
let currentActiveTab = 0; 
let currentLastUpdate = "";

function openTab(event, index) {
    currentActiveTab = index; 
    document.querySelectorAll('.tab-panel').forEach(panel => panel.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
    
    document.getElementById(`league-${index}`).classList.add('active');
    if(event) {
        event.currentTarget.classList.add('active');
    } else {
        let btns = document.querySelectorAll('.tab-btn');
        if(btns.length > 0) btns[index].classList.add('active');
    }
}

function fetchLiveScore() {
    fetch('livescore_data.json?t=' + new Date().getTime())
        .then(response => response.json())
        .then(data => {
            if (data.last_update === currentLastUpdate) return; 
            currentLastUpdate = data.last_update;

            document.getElementById('last-update').innerText = "Cập nhật lần cuối : " + data.last_update;
            
            if(!data.leagues || data.leagues.length === 0) {
                document.getElementById('tabs-container').innerHTML = '<p style="color:#ef4444; text-align:center; width:100%; grid-column: 1 / -1;">Dữ liệu đang trống...</p>';
                return;
            }

            // --- THUẬT TOÁN ĐOÁN NGÀY THÔNG MINH ---
            let todayStr = "";
            let tomorrowStr = "";
            let isEveningUpdate = false;
            
            try {
                // Tách thời gian từ "03/04/2026 18:50:10"
                let dateTimeParts = data.last_update.split(' ');
                let dateParts = dateTimeParts[0].split('/');
                let timeParts = dateTimeParts[1].split(':');
                
                let d = parseInt(dateParts[0]);
                let m = parseInt(dateParts[1]);
                let y = dateParts.length > 2 ? parseInt(dateParts[2]) : new Date().getFullYear();
                
                let updateDate = new Date(y, m - 1, d);
                let tomorrowDate = new Date(y, m - 1, d + 1); // Cộng thêm 1 ngày
                
                todayStr = String(updateDate.getDate()).padStart(2, '0') + '/' + String(updateDate.getMonth() + 1).padStart(2, '0');
                tomorrowStr = String(tomorrowDate.getDate()).padStart(2, '0') + '/' + String(tomorrowDate.getMonth() + 1).padStart(2, '0');
                
                // Cờ kiểm tra xem web cập nhật vào chiều tối hay không (>= 12h trưa)
                isEveningUpdate = parseInt(timeParts[0]) >= 12; 
            } catch(e) {
                // Đề phòng last_update rỗng hoặc lỗi format thì lấy ngày máy tính
                let now = new Date();
                let tmr = new Date(now.getTime() + 86400000);
                todayStr = String(now.getDate()).padStart(2, '0') + '/' + String(now.getMonth() + 1).padStart(2, '0');
                tomorrowStr = String(tmr.getDate()).padStart(2, '0') + '/' + String(tmr.getMonth() + 1).padStart(2, '0');
                isEveningUpdate = now.getHours() >= 12;
            }

            let tabsHTML = '';
            let contentHTML = '';

            data.leagues.forEach((league, index) => {
                let isActive = (index === currentActiveTab) ? 'active' : '';
                if(currentActiveTab >= data.leagues.length) {
                    isActive = (index === 0) ? 'active' : '';
                    currentActiveTab = 0;
                }
                
                let isTodayTab = league.league_name.toLowerCase().includes('hôm nay');
                
                tabsHTML += `<button class="tab-btn ${isActive}" onclick="openTab(event, ${index})" title="${league.league_name}">${league.league_name}</button>`;
                contentHTML += `<div id="league-${index}" class="tab-panel ${isActive}">`;
                
                league.days.forEach(day => {
                    contentHTML += `<div class="date-header">${day.date}</div>`;
                    
                    day.matches.forEach(match => {
                        // Cập nhật: Bắt thêm các trạng thái 'FT', 'Finished' đề phòng API thay đổi từ khóa
                        let scoreClass = (match.status === 'played' || match.status === 'FT' || match.status === 'Finished') ? 'played' : '';
                        
                        let timeHtml = '';
                        
                        // 1. Tách ngày và giờ nếu API trả về gộp
                        let extractedTime = match.time || ""; 
                        let extractedDate = match.date || "";

                        if (extractedTime.includes('/') || extractedTime.includes('<br>')) {
                            let parts = extractedTime.replace(/<br\s*\/?>|\n/gi, ' ').split(' ').filter(p => p.trim() !== '');
                            if (parts.length >= 2) {
                                extractedTime = parts[0];
                                extractedDate = parts[1];
                            }
                        }

                        // 2. Xử lý hiển thị thời gian và ngày dự đoán
                        if (isTodayTab) {
                            let matchDateDisplay = extractedDate;
                            if (!matchDateDisplay) {
                                let matchHour = parseInt(extractedTime.split(':')[0]);
                                if (isEveningUpdate && matchHour >= 0 && matchHour <= 11) {
                                    matchDateDisplay = tomorrowStr;
                                } else {
                                    matchDateDisplay = todayStr;
                                }
                            }
                            timeHtml = `
                                <span class="time-date">${matchDateDisplay}</span>
                                <span class="time-time">${extractedTime}</span>
                            `;
                        } else {
                            if (extractedDate) {
                                timeHtml = `
                                    <span class="time-date">${extractedDate}</span>
                                    <span class="time-time">${extractedTime}</span>
                                `;
                            } else {
                                timeHtml = `<span class="time-time">${extractedTime}</span>`;
                            }
                        }
                        
                        // Xử lý fallback cho tỉ số: nếu API không trả về score nhưng trận đã đá xong thì để '?', chưa đá thì 'vs'
                        let matchScore = match.score ? match.score : (scoreClass !== '' ? '?' : 'vs');

                        // 3. Render khối HTML cho từng trận đấu (KHÔNG ĐƯỢC XÓA PHẦN NÀY)
                        contentHTML += `
                            <div class="match-row ${scoreClass}">
                                <div class="time">${timeHtml}</div>
                                <div class="team home">${match.team_home}</div>
                                <div class="score-box"><span class="score">${matchScore}</span></div>
                                <div class="team away">${match.team_away}</div>
                            </div>
                        `;
                    });
                });
                contentHTML += `</div>`; 
            });
            
            document.getElementById('tabs-container').innerHTML = tabsHTML;
            document.getElementById('tab-content-container').innerHTML = contentHTML;
        })
        .catch(console.error);
}

fetchLiveScore();
setInterval(fetchLiveScore, 60000); 
</script>
