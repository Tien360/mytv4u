    function updateGmt7Clock() {
        const clockEl = document.getElementById('liveGmt7Clock');
        if (!clockEl) return;
        
        // Tạo đối tượng Date dựa trên mốc thời gian của server
        const now = new Date(serverTimeMs);
        
        // Cấu hình định dạng giờ theo múi giờ GMT+7
        const options = {

        };

        serverTimeMs += 1000;
